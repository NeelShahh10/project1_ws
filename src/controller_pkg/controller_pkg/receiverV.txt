#!/usr/bin/env python3
import rclpy
from rclpy.executors import ExternalShutdownException
from rclpy.node import Node
from geometry_msgs.msg import Quaternion, Twist

TIMEOUT = 0.5          # seconds without a command -> stop
LINEAR_SCALE = 2.0     # turtle speed for full stick
ANGULAR_SCALE = 2.0


class TurtleSafety(Node):
    def __init__(self):
        super().__init__('turtle_safety')
        self.create_subscription(Quaternion, 'cmd_in', self.cb, 10)
        self.pub = self.create_publisher(Twist, 'turtle1/cmd_vel', 10)
        self.last = None
        self.cmd = Twist()
        self.stopped = False
        self.create_timer(0.05, self.loop)
        self.get_logger().info('turtle_safety ready')

    def cb(self, q):
        t = Twist()
        if q.z > 0.5:                      # RB held
            t.linear.x = q.y * LINEAR_SCALE
            t.angular.z = -q.x * ANGULAR_SCALE
        self.cmd = t
        self.last = self.get_clock().now()

    def loop(self):
        if self.last is None:
            return
        age = (self.get_clock().now() - self.last).nanoseconds / 1e9
        if age > TIMEOUT:
            if not self.stopped:
                self.stopped = True
                self.get_logger().warn('No command for 500 ms - STOPPING TURTLE')
            self.pub.publish(Twist())
        else:
            if self.stopped:
                self.stopped = False
                self.get_logger().info('Commands received again')
            self.pub.publish(self.cmd)


def main(args=None):
    rclpy.init(args=args)
    node = TurtleSafety()
    try:
        rclpy.spin(node)
    except (KeyboardInterrupt, ExternalShutdownException):
        pass
    finally:
        node.destroy_node()
        if rclpy.ok():
            rclpy.shutdown()


if __name__ == '__main__':
    main()
